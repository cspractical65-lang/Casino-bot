import telebot

import os
import time

BOT_TOKEN = "YOUR_BOT_TOKEN_HERE"
ADMIN_IDS = [YOUR_ADMIN_ID_HERE]  # e.g., [123456789, 987654321]

bot = telebot.TeleBot(BOT_TOKEN)

# Data files
USERS_FILE = "casino_users.json"
GAMES_FILE = "games_history.json"
TRANSACTIONS_FILE = "transactions.json"
ADMIN_LOG_FILE = "admin_log.json"
AIRDROP_LOG_FILE = "airdrop_log.json"

def load_data(filename):
    if os.path.exists(filename):
        with open(filename, "r") as f:
            return json.load(f)
    return {}

def save_data(data, filename):
    with open(filename, "w") as f:
        json.dump(data, f, indent=2)

users = load_data(USERS_FILE)
games = load_data(GAMES_FILE)
transactions = load_data(TRANSACTIONS_FILE)
admin_log = load_data(ADMIN_LOG_FILE)
airdrop_log = load_data(AIRDROP_LOG_FILE)

def get_user(user_id):
    user_id = str(user_id)
    if user_id not in users:
        users[user_id] = {
            "balance": Decimal("1000"),
            "total_deposited": Decimal("0"),
            "total_withdrawn": Decimal("0"),
            "total_wagered": Decimal("0"),
            "total_won": Decimal("0"),
            "total_lost": Decimal("0"),
            "games_played": 0,
            "wallet": None,
            "vip": False,
            "banned": False,
            "created_at": datetime.now().isoformat(),
            "referrer": None,
            "referrals": 0,
            "last_rain": None,
            "last_airdrop": None
        }
    return users[user_id]

def update_user(user_id, **kwargs):
    user_id = str(user_id)
    for key, value in kwargs.items():
        users[user_id][key] = value
    save_data(users, USERS_FILE)

def log_game(user_id, game_type, bet, result, payout, multiplier=1):
    game_id = str(int(time.time() * 1000))
    games[game_id] = {
        "user": str(user_id),
        "game": game_type,
        "bet": str(bet),
        "result": result,
        "payout": str(payout),
        "multiplier": multiplier,
        "net": str(Decimal(payout) - Decimal(bet)),
        "time": datetime.now().isoformat()
    }
    save_data(games, GAMES_FILE)

def log_transaction(user_id, txn_type, amount, status, details=""):
    txn_id = str(int(time.time() * 1000))
    transactions[txn_id] = {
        "user": str(user_id),
        "type": txn_type,
        "amount": str(amount),
        "status": status,
        "details": details,
        "time": datetime.now().isoformat()
    }
    save_data(transactions, TRANSACTIONS_FILE)

def is_admin(user_id):
    return user_id in ADMIN_IDS

def log_admin_action(admin_id, action, details):
    log_id = str(int(time.time() * 1000))
    admin_log[log_id] = {
        "admin": admin_id,
        "action": action,
        "details": details,
        "time": datetime.now().isoformat()
    }
    save_data(admin_log, ADMIN_LOG_FILE)

def play_dice(user_id, bet):
    user = get_user(user_id)
    bet = Decimal(str(bet))
    
    if user["banned"]:
        return "❌ Account banned", None
    
    if bet > Decimal(user["balance"]):
        return "❌ Low balance", None
    
    player_roll = random.randint(1, 100)
    dealer_roll = random.randint(1, 100)
    
    if player_roll > dealer_roll:
        result = "WIN"
        multiplier = 1.98
        payout = bet * Decimal("1.98")
    elif player_roll == dealer_roll:
        result = "PUSH"
        payout = bet
    else:
        result = "LOSE"
        payout = Decimal("0")
    
    new_balance = Decimal(user["balance"]) - bet + payout
    update_user(user_id,
        balance=new_balance,
        total_wagered=Decimal(user["total_wagered"]) + bet,
        total_won=Decimal(user["total_won"]) + payout if result == "WIN" else Decimal(user["total_won"]),
        total_lost=Decimal(user["total_lost"]) + bet if result == "LOSE" else Decimal(user["total_lost"]),
        games_played=user["games_played"] + 1
    )
    
    log_game(user_id, "dice", bet, f"You:{player_roll} vs Dealer:{dealer_roll}", payout, multiplier if result == "WIN" else 1)
    
    if result == "WIN":
        msg = f"""🎲 **DICE - WIN!**

You: {player_roll}
Dealer: {dealer_roll}
Multiplier: 1.98x
Win: ${payout}

Balance: ${new_balance}
"""
    elif result == "PUSH":
        msg = f"""🎲 **DICE - PUSH**

You: {player_roll}
Dealer: {dealer_roll}
Push: ${bet}

Balance: ${new_balance}
"""
    else:
        msg = f"""🎲 **DICE - LOSE**

You: {player_roll}
Dealer: {dealer_roll}
Lost: ${bet}

Balance: ${new_balance}
"""
    
    return msg, result

@bot.callback_query_handler(func=lambda call: call.data == "send_airdrop")
def send_airdrop_handler(call):
    if not is_admin(call.message.chat.id):
        return
    
    msg = bot.send_message(call.message.chat.id, "Enter airdrop amount per user ($):")
    bot.register_next_step_handler(msg, process_airdrop, call.message.chat.id)

def process_airdrop(message, admin_id):
    try:
        amount = Decimal(message.text.strip())
        count = 0
        
        for user_id in users.keys():
            user = get_user(user_id)
            if not user["banned"]:
                new_balance = Decimal(user["balance"]) + amount
                update_user(user_id, balance=new_balance)
                count += 1
                
                try:
                    bot.send_message(int(user_id), f"🎁 **AIRDROP!**\n\nYou received ${amount}\n\nNew Balance: ${new_balance}")
                except:
                    pass
        
        airdrop_id = str(int(time.time() * 1000))
        airdrop_log[airdrop_id] = {
            "admin": admin_id,
            "amount": str(amount),
            "users": count,
            "time": datetime.now().isoformat()
        }
        save_data(airdrop_log, AIRDROP_LOG_FILE)
        log_admin_action(admin_id, "airdrop", f"${amount} to {count} users")
        
        bot.send_message(admin_id, f"✅ Airdrop sent to {count} users\nAmount per user: ${amount}")
    except:
        bot.send_message(admin_id, "❌ Invalid amount")

@bot.message_handler(commands=['rain'])
def rain_menu(message):
    if not is_admin(message.chat.id):
        bot.send_message(message.chat.id, "❌ Admin only")
        return
    
    msg = bot.send_message(message.chat.id, "Enter total rain amount ($) to distribute:")
    bot.register_next_step_handler(msg, process_rain, message.chat.id)

def process_rain(message, admin_id):
    try:
        total_amount = Decimal(message.text.strip())
        
        # Get active users (played in last 24h)
        active_users = []
        for user_id in users.keys():
            user = get_user(user_id)
            if not user["banned"] and user["games_played"] > 0:
                active_users.append(user_id)
        
        if not active_users:
            bot.send_message(admin_id, "❌ No active users")
            return
        
        # Distribute equally
        per_user = total_amount / Decimal(len(active_users))
        
        for user_id in active_users:
            user = get_user(user_id)
            new_balance = Decimal(user["balance"]) + per_user
            update_user(user_id, balance=new_balance, last_rain=datetime.now().isoformat())
            
            try:
                bot.send_message(int(user_id), f"🌧️ **RAIN!**\n\nYou received ${per_user:.2f}\n\nNew Balance: ${new_balance:.2f}")
            except:
                pass
        
        log_admin_action(admin_id, "rain", f"${total_amount} to {len(active_users)} users (${per_user:.2f} each)")
        bot.send_message(admin_id, f"✅ Rain sent!\n\nTotal: ${total_amount}\nUsers: {len(active_users)}\nPer user: ${per_user:.2f}")
    except:
        bot.send_message(admin_id, "❌ Invalid amount")

    bot.edit_message_text(text, user_id, call.message.message_id, reply_markup=markup, parse_mode="Markdown")

@bot.callback_query_handler(func=lambda call: "_bet_" in call.data)
def handle_bet(call):
    user_id = call.message.chat.id
    user = get_user(user_id)
    
    if user["banned"]:
        bot.answer_callback_query(call.id, "❌ Banned", show_alert=True)
        return
    
    parts = call.data.split("_")
    game = parts[0]
    bet = int(parts[2])
    
    if game == "dice":
        msg, result = play_dice(user_id, bet)
    elif game == "predict":
        msg, result = play_predict(user_id, bet)
    elif game == "crash":
        msg, result = play_crash(user_id, bet)
    elif game == "plinko":
        msg, result = play_plinko(user_id, bet)
    
    user = get_user(user_id)
    markup = types.InlineKeyboardMarkup()
    markup.add(types.InlineKeyboardButton("🎮 Play Again", callback_data="play_again"))
    
    bot.edit_message_text(msg, user_id, call.message.message_id, parse_mode="Markdown")
    bot.send_message(user_id, "Next?", reply_markup=markup)

@bot.callback_query_handler(func=lambda call: call.data == "play_again")

    play_menu(call.message)
    bot.answer_callback_query(call.id)
